function(gp100_add_test name source)
    set(target "gp100_${name}_test")
    add_executable("${target}" "${source}")
    target_link_libraries("${target}" PRIVATE gp100_core)
    target_compile_options("${target}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ARGN)
        add_dependencies("${target}" gp100_test_fixtures)
    endif()
    add_test(NAME "${name}" COMMAND "${target}" ${ARGN})
endfunction()
